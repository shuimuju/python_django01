### 传输:防止中间人攻击

> 1. 必须全程使用HTTPS(TLS加密)
>
> 2. 对API请求中携带验证码的地方，token/签名机制，防止重放攻击
>
> 3. 对图片验证码，可以在URL上加签名参数(HMAC)

```python
import hmac
import hashlib
import time
# 图片验证码 
SECRET_KEY = b"your_secret_key"

def generate_signed_url(code_id):
    timestamp = str(int(time.time()))
    msg = f"{code_id}:{timestamp}".encode()
    signature = hmac.new(SECRET_KEY, msg, hashlib.sha256).hexdigest()
    return f"/captcha/{code_id}?ts={timestamp}&sign={signature}"

print(generate_signed_url("abc123"))
```

### 校验

> 1. 限制错误次数(同一IP或用户5次失败就锁定)
> 2. 验证码一次性使用(验证成功就删除)
> 3. 验证码的时效性(2-5分钟)

```python
class CodeManager:
    def __init__(self):
        self.codes ={}
        self.salt = os.urandom(16)
    def set_code(self,user,code):
        self.codes[user] = {
            'code': hashlib.pbkdf2_hmac('sha256',code.encode(),self.salt,100_000),
            'expires_at':datetime.now() + timedelta(minutes=2),
            'attempts':0
        }
    def verify_code(self,user,input_code):
        input_code = hashlib.pbkdf2_hmac('sha256',input_code.encode(),self.salt,100_000)
        print(input_code)
        data_user = self.codes.get(user)
        if not data_user:
            return False
        if datetime.now() > data_user['expires_at']:
            return False
        if data_user['attempts'] >=5:
            return False
        data_user['attempts'] = data_user['attempts'] + 1
        if input_code == data_user['code']:
            del self.codes[user]
            return True
def index40():
    digits = string.digits
    code = ''.join(secrets.choice(digits) for _ in range(6))
    print(code)
    code_manager = CodeManager()
    code_manager.set_code('1123',code)
    print(code_manager.verify_code('1123',code))
```

### 图形验证码:captcha,Pillow

#### 后端生成

```python
import secrets
import string
from io import BytesIO
import random
from PIL import Image, ImageFont, ImageDraw,ImageFilter
from django.http import HttpResponse
from django.shortcuts import render

def create_captcha_image(request):
    # 验证码内容形式
    code = string.ascii_letters + string.digits
    captcha_text = ''.join(secrets.choice(code) for i in range(6))
    width, height = 130 , 40
    # 创建图片
    image = Image.new('RGB', (width, height), color='white')
    # 创建画笔
    draw = ImageDraw.Draw(image)
    # 字体
    try:
        font = ImageFont.truetype('/System/Library/Fonts/Arial.ttf',20)
    except:
        try:
            font = ImageFont.truetype('arial.ttf', 20)
        except:
            font = ImageFont.load_default()
    # 绘制验证码文字
    colors = ['#FF0000', '#00FF00', '#0000FF', '#FF00FF', '#FFA500']
    for i, char in enumerate(captcha_text):
        x = 10 + i * 20
        y = random.randint(5,15)
        color = random.choice(colors)
        draw.text((x, y), char, font=font, fill=color)
    # 添加噪点
    for _ in range(200):
        x = random.randint(0, width)
        y = random.randint(0, height)
        draw.point((x, y), fill=random.choice(['#000000','#999999']))
    # 添加干扰线
    for _ in range(3):
        x1 = random.randint(0, width)
        y1 = random.randint(0, height)
        x2 = random.randint(0, width)
        y2 = random.randint(0, height)
        draw.line([(x1,y1),(x2,y2)],fill='#CCCCCC',width=1)
    # 应用滤镜模糊
    image = image.filter(ImageFilter.GaussianBlur(radius=0.5))
    buffer = BytesIO()
    image.save(buffer, 'PNG')
    buffer.seek(0)
    return HttpResponse(buffer.getvalue(), content_type='image/png')
```

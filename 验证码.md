### 传输:防止中间人攻击

> 1. 必须全程使用HTTPS(TLS加密)
>
> 2. 对API请求中携带验证码的地方，陪客token/签名机制，防止重放攻击
>
> 3. 对图片验证码，可以在URL上加签名参数(HMAC)

```python
import hmac
import hashlib
import time
# 图片验证码 
SECRET_KEY = b"your_secret_key"

def generate_signed_url(code_id):
    timestamp = str(int(time.time()))
    msg = f"{code_id}:{timestamp}".encode()
    signature = hmac.new(SECRET_KEY, msg, hashlib.sha256).hexdigest()
    return f"/captcha/{code_id}?ts={timestamp}&sign={signature}"

print(generate_signed_url("abc123"))
```

### 校验

> 1. 限制错误次数(同一IP或用户5次失败就锁定)
> 2. 验证码一次性使用(验证成功就删除)
> 3. 验证码的时效性(2-5分钟)

```python
class CodeManager:
    def __init__(self):
        self.codes ={}
        self.salt = os.urandom(16)
    def set_code(self,user,code):
        self.codes[user] = {
            'code': hashlib.pbkdf2_hmac('sha256',code.encode(),self.salt,100_000),
            'expires_at':datetime.now() + timedelta(minutes=2),
            'attempts':0
        }
    def verify_code(self,user,input_code):
        input_code = hashlib.pbkdf2_hmac('sha256',input_code.encode(),self.salt,100_000)
        print(input_code)
        data_user = self.codes.get(user)
        if not data_user:
            return False
        if datetime.now() > data_user['expires_at']:
            return False
        if data_user['attempts'] >=5:
            return False
        data_user['attempts'] = data_user['attempts'] + 1
        if input_code == data_user['code']:
            del self.codes[user]
            return True
def index40():
    digits = string.digits
    code = ''.join(secrets.choice(digits) for _ in range(6))
    print(code)
    code_manager = CodeManager()
    code_manager.set_code('1123',code)
    print(code_manager.verify_code('1123',code))
```
